cmake_minimum_required(VERSION 3.16)
project(promax_voice_dsp_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(promax_voice_dsp STATIC PromaxVoiceDsp.cpp)
target_include_directories(promax_voice_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(promax_voice_dsp PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(promax_voice_dsp_test tests/PromaxVoiceDspTest.cpp)
target_link_libraries(promax_voice_dsp_test PRIVATE promax_voice_dsp)
target_compile_options(promax_voice_dsp_test PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_test(NAME promax_voice_dsp_test COMMAND promax_voice_dsp_test)
